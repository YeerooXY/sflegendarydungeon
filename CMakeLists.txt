cmake_minimum_required(VERSION 3.20)
project(sf_legendary_sim VERSION 0.6.12 LANGUAGES CXX)

option(SFLD_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(sfld_core src/model.cpp src/profile.cpp src/progress.cpp src/engine.cpp src/policy.cpp src/strategy.cpp src/adaptive_gems.cpp src/batch.cpp src/experiment.cpp src/experiment_cli.cpp src/rollout.cpp)
target_include_directories(sfld_core PUBLIC include)
target_compile_features(sfld_core PUBLIC cxx_std_20)
target_link_libraries(sfld_core PUBLIC Threads::Threads)
set_target_properties(sfld_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(sfld_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sfld_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(SFLD_SANITIZERS AND NOT MSVC)
  target_compile_options(sfld_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sfld_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sfld src/main.cpp)
target_link_libraries(sfld PRIVATE sfld_core)
add_library(sfld_rules_core src/rule_gems.cpp src/rule_policy.cpp)
target_link_libraries(sfld_rules_core PUBLIC sfld_core)
set_target_properties(sfld_rules_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(sfld_rules_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sfld_rules_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(sfld_rule_search tools/rule_search.cpp)
target_link_libraries(sfld_rule_search PRIVATE sfld_rules_core)
add_executable(sfld_rollout tools/rollout_advisor.cpp)
target_link_libraries(sfld_rollout PRIVATE sfld_rules_core)
add_library(sfld_rl SHARED src/rl_env.cpp)
target_link_libraries(sfld_rl PRIVATE sfld_rules_core)
target_include_directories(sfld_rl PUBLIC include)
target_compile_features(sfld_rl PUBLIC cxx_std_20)
set_target_properties(sfld_rl PROPERTIES CXX_EXTENSIONS OFF)
# The visual/desktop host launches sfld and loads this sibling library for
# final-stone continuations. A targeted sfld build must produce both artifacts.
add_dependencies(sfld sfld_rl)
