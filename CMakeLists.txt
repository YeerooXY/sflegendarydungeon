cmake_minimum_required(VERSION 3.20)
project(sf_legendary_sim VERSION 0.2.0 LANGUAGES CXX)

option(SFLD_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(sfld_core src/model.cpp src/profile.cpp src/progress.cpp src/engine.cpp src/policy.cpp src/batch.cpp)
target_include_directories(sfld_core PUBLIC include)
target_compile_features(sfld_core PUBLIC cxx_std_20)
target_link_libraries(sfld_core PUBLIC Threads::Threads)
set_target_properties(sfld_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(sfld_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sfld_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(SFLD_SANITIZERS AND NOT MSVC)
  target_compile_options(sfld_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sfld_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sfld src/main.cpp)
target_link_libraries(sfld PRIVATE sfld_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(sfld_tests tests/tests.cpp)
  target_link_libraries(sfld_tests PRIVATE sfld_core)
  target_compile_definitions(sfld_tests PRIVATE SFLD_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME mechanics COMMAND sfld_tests)
  add_test(NAME audit COMMAND sfld audit --profile ${CMAKE_CURRENT_SOURCE_DIR}/profiles/synthetic.profile)
  add_test(NAME assumptions_required COMMAND sfld simulate --profile ${CMAKE_CURRENT_SOURCE_DIR}/profiles/synthetic.profile --runs 2)
  set_tests_properties(assumptions_required PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli COMMAND ${CMAKE_COMMAND} -DSFLD=$<TARGET_FILE:sfld>
    -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR} -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/cli-test
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
endif()
install(TARGETS sfld RUNTIME DESTINATION bin)
install(DIRECTORY profiles examples docs DESTINATION share/sfld)
